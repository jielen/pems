```java
@Service
public class PemsAuditLogServiceImpl implements IPemsAuditLogService {

    @Autowired
    private PemsAuditLogMapper auditLogMapper;

    // HMAC secret key - should be in config/env
    private static final String HMAC_KEY = "pems-audit-secret-key";

    @Override
    public void recordLog(Long evidenceId, String operationType, String detail) {
        PemsAuditLog log = new PemsAuditLog();
        log.setEvidenceId(evidenceId);
        log.setOperationType(operationType);
        log.setDetail(detail);
        // Get current user from SecurityUtils
        // Set operatorId, operatorName, operatorBadge, operatorUnit, operatorIp, operateTime

        // Get previous hash for chain
        PemsAuditLog prevLog = auditLogMapper.selectLatest();
        String prevHash = (prevLog != null) ? prevLog.getHashValue() : "GENESIS";
        log.setPrevHash(prevHash);

        // Compute current hash: HMAC-SHA256(prevHash + evidenceId + operationType + operatorId + operateTime + detail)
        String hash = computeHash(log, prevHash);
        log.setHashValue(hash);

        auditLogMapper.insert(log);  // ONLY insert, immutable
    }

    @Override
    public String computeHash(PemsAuditLog log, String prevHash) {
        String data = prevHash + "|" +
                      log.getEvidenceId() + "|" +
                      log.getOperationType() + "|" +
                      log.getOperatorId() + "|" +
                      log.getOperateTime() + "|" +
                      log.getDetail();
        try {
            Mac mac = Mac.getInstance("HmacSHA256");
            SecretKeySpec keySpec = new SecretKeySpec(HMAC_KEY.getBytes(), "HmacSHA256");
            mac.init(keySpec);
            byte[] hash = mac.doFinal(data.getBytes());
            return bytesToHex(hash);
        } catch (Exception e) {
            throw new RuntimeException("Hash computation failed", e);
        }
    }

    @Override
    public int verifyIntegrity() {
```

```java
@RestController
@RequestMapping("/pems/audit/log")
public class PemsAuditLogController {

    @PreAuthorize("@ss.hasPermi('pems:log:list')")   // Auditor role
    @GetMapping("/list")
    public TableDataInfo list(PemsAuditLog query) {
        startPage();
        List<PemsAuditLog> list = auditLogService.selectLogList(query);
        return getDataTable(list);
    }

    @PreAuthorize("@ss.hasPermi('pems:log:export')")
    @GetMapping("/export")
    public AjaxResult export(PemsAuditLog query) {
        List<PemsAuditLog> list = auditLogService.selectLogList(query);
        // Use POI to export Excel
        // Columns: time, operator, badge, unit, operation, detail, hash
        return ExcelUtil.export(list, "audit_log");
    }

    @PreAuthorize("@ss.hasPermi('pems:log:list')")
    @GetMapping("/verify")
    public AjaxResult verifyIntegrity() {
        int broken = auditLogService.verifyIntegrity();
        if (broken == 0) {
            return success("Hash chain valid");
        } else {
            return error("Hash chain broken at " + broken + " position(s)");
        }
    }
}
```
